#define BUTTON_1 3

#define GREEN_LED 12

#define SEGMENT_A 4

int cathode = 0;
int anode = 1;

void setup()
{
  for(int i = 4; i<11; i++)
  {
   	pinMode(i, OUTPUT);
    
   	digitalWrite(i, HIGH); // ha anodós a kijelző
    //digitalWrite(i, LOW); // ha katódos a kijelző
  }

  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  
  pinMode(BUTTON_1, INPUT);
  pinMode(BUTTON_2, INPUT);

  Serial.begin(9600);
}

void loop()
{
  if(digitalRead(BUTTON_1))
  {
	digitalWrite(RED_LED, HIGH);
    digitalWrite(GREEN_LED, LOW);
    Display(0, anode);

    delay(100);
  }
  if(digitalRead(BUTTON_2))
  {
	digitalWrite(GREEN_LED, HIGH);
    digitalWrite(RED_LED, LOW);
    Display(8, anode);

    delay(100);
  }
}
