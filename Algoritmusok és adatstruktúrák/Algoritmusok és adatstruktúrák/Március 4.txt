#include <stdio.h>
#include <stdlib.h>

int main()
{
    int n = 0;
    int i = 0;

    int minValue = 0;
    int maxValue = 100;

    int amount = 0;

    int input = 0;

    double avrage = 0;

    int maxNValue = 0;
    int minNValue = 0;

    int negativeNumberCounter = 0;
    int negativeNumberAmount = 0;

    int positiveNumberCounter = 0;
    int positiveNumbermultiply = 1;

    printf("Kerem n erteket:");
    scanf("%d", &n);

    if(n < maxValue && n > minValue)
    {
        printf("Adjon meg n darab szamot!");

        for(i = 0; i < n; i++)
        {
            scanf("%d", &input);

            amount = amount + input;

            if(i == 0)
            {
                minNValue = input;
                maxNValue = input;
            }

            if(input > maxNValue)
            {
                maxNValue = input;
            }

            if(input < minNValue)
            {
                minNValue = input;
            }

            if(input < 0)
            {
                negativeNumberCounter ++;
                negativeNumberAmount = negativeNumberAmount + input;

                if(negativeNumberCounter == n)
                {
                    positiveNumbermultiply = 0;
                }
            }

            if(input > 0)
            {
                positiveNumberCounter ++;
                positiveNumbermultiply = positiveNumbermultiply * input;
            }
        }

        avrage = (double)amount / n;
    }

    printf("Osszeg:");
    printf("%d", amount);
    printf("\n");

    printf("atlag:");
    printf("%f", avrage);
    printf("\n");

    printf("max:");
    printf("%d", maxNValue);
    printf("\n");

    printf("min:");
    printf("%d", minNValue);
    printf("\n");


    printf("negativ szamok osszege:");
    printf("%d", negativeNumberAmount);
    printf("\n");

    printf("negativ szamok darabszama:");
    printf("%d", negativeNumberCounter);
    printf("\n");

    printf("pozitiv szamok darabszama:");
    printf("%d", positiveNumberCounter);
    printf("\n");

    printf("pozitiv szamok szorzata:");
    printf("%d", positiveNumbermultiply);
    printf("\n");

    return 0;
}
